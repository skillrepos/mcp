"""
agent_mcp.py - a minimal MCP agent, no agent framework involved.

Updated 2026-08-03 for MCP specification revision 2026-07-28.

Why there is no LangChain here anymore:
  langchain-mcp-adapters imports mcp.server.fastmcp, a module removed in MCP
  Python SDK v2. It cannot be installed alongside a 2026-07-28-capable stack.
  Writing the loop by hand is only ~40 lines and shows you exactly what an
  "agent" is: ask the model, run whatever tool it names, feed the result back,
  repeat. Every framework you will ever use is a wrapper around this loop.

The loop:
  1. tools/list  -> ask the MCP server what it can do
  2. translate the MCP tool schemas into the LLM's tool-calling format
  3. send the question + tool list to the model
  4. the model answers with tool calls -> run them via tools/call
  5. feed results back and repeat until the model answers in plain text
"""

import asyncio
import json

import httpx

from fastmcp import Client

MCP_URL = "http://127.0.0.1:8931/mcp"
OLLAMA_URL = "http://127.0.0.1:11434/api/chat"
MODEL = "llama3.2"

BLUE = "\033[94m"
GREEN = "\033[92m"
YELLOW = "\033[93m"
RESET = "\033[0m"

SYSTEM = (
    "You are a careful calculator assistant. Use the provided tools for every "
    "arithmetic step. Do exactly one operation per tool call and never nest "
    "calls. After you have the final number, state it in one short sentence."
)


def to_ollama_tools(mcp_tools):
    """Translate MCP tool definitions into Ollama's tool-calling format.

    This is the M x N problem in miniature, and the reason MCP exists: the
    tool's schema arrived from the server in a standard shape, and we adapt it
    once, here, for whichever model vendor we happen to be using.
    """
    out = []
    for t in mcp_tools:
        out.append(
            {
                "type": "function",
                "function": {
                    "name": t.name,
                    "description": t.description or "",
                    # MCP hands us JSON Schema, which is exactly what the model
                    # APIs want. We never hand-wrote a schema.
                    "parameters": t.input_schema,
                },
            }
        )
    return out


async def ask_model(client: httpx.AsyncClient, messages, tools):
    """One turn with the LLM. Retries because cold local models are slow."""
    payload = {"model": MODEL, "messages": messages, "tools": tools, "stream": False}
    for attempt in range(4):
        try:
            resp = await client.post(OLLAMA_URL, json=payload)
            resp.raise_for_status()
            return resp.json()["message"]
        except (httpx.TimeoutException, httpx.ConnectError):
            if attempt == 3:
                raise
            await asyncio.sleep(1.5 * (attempt + 1))
    raise RuntimeError("model unreachable")


async def main():
    timeout = httpx.Timeout(connect=30.0, read=300.0, write=30.0, pool=30.0)

    # 1) Connect to the MCP server. mode="auto" (the default) speaks
    #    2026-07-28 when the server does, and falls back to the older
    #    handshake era when it does not.
    async with Client(MCP_URL) as mcp, httpx.AsyncClient(timeout=timeout) as http:
        print(f"{BLUE}Connected using protocol {mcp.protocol_version}{RESET}")

        # 2) Discovery: the server tells us what it offers. We wrote no
        #    endpoint URLs, no query strings, and no JSON schemas.
        mcp_tools = await mcp.list_tools()
        print(f"{BLUE}Discovered tools: {[t.name for t in mcp_tools]}{RESET}\n")
        tools = to_ollama_tools(mcp_tools)

        messages = [
            {"role": "system", "content": SYSTEM},
            {"role": "user", "content": "What is 12 x 8 / 3 ?"},
        ]

        # 3) The agent loop.
        for turn in range(8):
            message = await ask_model(http, messages, tools)
            messages.append(message)

            calls = message.get("tool_calls") or []
            if not calls:
                break  # a plain-text answer means the model is done

            # 4) The model may request several tools in one turn. Run them in
            #    order and append each result before going back to the model.
            for call in calls:
                fn = call["function"]
                name = fn["name"]
                args = fn.get("arguments") or {}
                if isinstance(args, str):
                    args = json.loads(args)

                print(f"{YELLOW}-> tools/call {name}({args}){RESET}")
                result = await mcp.call_tool(name, args)

                # FastMCP 4 returns a CallToolResult object. .data is the
                # hydrated Python value; .content holds raw content blocks.
                value = result.data
                if value is None and result.content:
                    value = result.content[0].text
                print(f"{GREEN}<- {value}{RESET}")

                messages.append(
                    {"role": "tool", "tool_name": name, "content": str(value)}
                )

        # 5) Show the trace and the final answer.
        print(f"\n{BLUE}=== FULL AGENT TRACE ==={RESET}")
        for m in messages:
            role = m.get("role")
            if role == "system":
                continue
            if role == "tool":
                print(f"TOOL RESULT ({m.get('tool_name')}): {m.get('content')}")
            elif m.get("tool_calls"):
                names = [c["function"]["name"] for c in m["tool_calls"]]
                print(f"ASSISTANT: (requested tools: {', '.join(names)})")
            elif m.get("content"):
                print(f"{role.upper()}: {m['content']}")

        final = ""
        for m in reversed(messages):
            if m.get("role") == "assistant" and (m.get("content") or "").strip():
                final = m["content"].strip()
                break

        print(f"\n{GREEN}=== FINAL ANSWER ==={RESET}")
        print(final)


if __name__ == "__main__":
    asyncio.run(main())
