"""
agent_mcp.py - a minimal MCP agent, no agent framework involved.

An "agent" is just this loop:
  1. tools/list  -> ask the MCP server what it can do
  2. translate the MCP tool schemas into the LLM's tool-calling format
  3. send the question + tool list to the model
  4. the model answers with tool calls -> run them via tools/call
  5. feed results back and repeat until the model answers in plain text

The HTTP plumbing and trace printing live in agent_helpers.py, so everything
below is MCP.
"""

import asyncio
import json

import httpx

from fastmcp import Client

from agent_helpers import (
    BLUE, GREEN, RESET, TIMEOUT, YELLOW,
    ask_model, print_trace, resolve_placeholders, to_number,
)

MCP_URL = "http://127.0.0.1:8931/mcp"

MAX_TURNS = 8  # stops a confused model from looping forever

SYSTEM = (
    "You are a careful calculator assistant. Use the provided tools for every "
    "arithmetic step. Do exactly one operation per tool call and never nest "
    "calls. When a step depends on an earlier tool result, pass the actual "
    "number that tool returned - never the word 'result' or any placeholder. "
    "After you have the final number, state it in one short sentence."
)


def to_ollama_tools(mcp_tools):
    """Translate MCP tool definitions into Ollama's tool-calling format.

    This is the M x N problem in miniature, and the reason MCP exists: the
    tool's schema arrived from the server in a standard shape, and we adapt it
    once, here, for whichever model vendor we happen to be using.
    """
    out = []
    for t in mcp_tools:
        out.append(
            {
                "type": "function",
                "function": {
                    "name": t.name,
                    "description": t.description or "",
                    # MCP hands us JSON Schema, which is exactly what the model
                    # APIs want. We never hand-wrote a schema.
                    "parameters": t.input_schema,
                },
            }
        )
    return out


async def main():
    # 1) Connect to the MCP server. mode="auto" (the default) speaks
    #    2026-07-28 when the server does, and falls back to the older
    #    handshake era when it does not.
    async with Client(MCP_URL) as mcp, httpx.AsyncClient(timeout=TIMEOUT) as http:
        print(f"{BLUE}Connected using protocol {mcp.protocol_version}{RESET}")

        # 2) Discovery: the server tells us what it offers. We wrote no
        #    endpoint URLs, no query strings, and no JSON schemas.
        mcp_tools = await mcp.list_tools()
        print(f"{BLUE}Discovered tools: {[t.name for t in mcp_tools]}{RESET}\n")
        tools = to_ollama_tools(mcp_tools)

        messages = [
            {"role": "system", "content": SYSTEM},
            {"role": "user", "content": "What is 12 x 8 / 3 ?"},
        ]

        # 3) The agent loop.
        last_value = None
        for turn in range(MAX_TURNS):
            message = await ask_model(http, messages, tools)
            messages.append(message)

            calls = message.get("tool_calls") or []
            if not calls:
                break  # a plain-text answer means the model is done

            # 4) Run each tool the model asked for, then feed the results back.
            for call in calls:
                name = call["function"]["name"]
                args = call["function"].get("arguments") or {}
                if isinstance(args, str):
                    args = json.loads(args)
                args = resolve_placeholders(args, last_value)

                print(f"{YELLOW}-> tools/call {name}({args}){RESET}")
                result = await mcp.call_tool(name, args)

                # .data is the hydrated Python value; .content holds raw blocks.
                value = result.data
                if value is None and result.content:
                    value = result.content[0].text
                print(f"{GREEN}<- {value}{RESET}")

                last_value = to_number(value, last_value)
                messages.append(
                    {"role": "tool", "tool_name": name, "content": str(value)}
                )

        # 5) Show the trace and the final answer.
        print_trace(messages)


if __name__ == "__main__":
    asyncio.run(main())
